# config/firewall.txt — the ROM audit's sources, in ONE file (installed by decomp-architect, install.py S2).
# The audit (tools/audit_public.py) derives everything it forbids from the lines below; it never carries a typed list.
#
#   purge:    <prefix-or-glob>   a path that may never be tracked (the rewrite's purge set reads the same lines, so the
#                                audit and any future history rewrite can never disagree about what is forbidden)
#   required: <path>             a hash source that MUST exist: a `.jsonl` manifest (one object per line with a `sha1`
#                                key; lines without one, e.g. coverage and `trivial` records, are skipped) or a checksum
#                                file (`<sha1>  <name>` lines, sha1sum format). Missing = the audit FAILS.
#   pending:  <path>             a hash source a later phase creates; missing = a loud warning, never a pass on its own.
#                                Promote it to `required:` in the phase that creates it (the phase ladder names which).
#   fixture:  <path>             the planted-fixture control's sha1 (install.py S2); counts as a required source.
#
# The audit refuses to run with zero resolvable hash sources or zero purge rules (a tool must refuse, never pass vacuously).

# ---- purge rules: the nine classes of the ROM firewall (mirror .gitignore's paths) ----
purge: disks/
# Disc-image formats are forbidden anywhere (fnmatch `*` crosses `/`). `*.bin` is NOT globbed: tracked fixtures, tests
# and build contracts use .bin names; the hash rule and the disks/, build/, dumps/**/*.bin rules cover real dumps.
glob:  *.cue
glob:  *.iso
glob:  *.img
glob:  *.chd
glob:  *.ecm
glob:  *.pbp
glob:  *.toc
# PCSX-Redux savestates (emulated RAM/VRAM = game bytes; tools/replay/README.md)
glob:  *.sstate
purge: SLUS_013.34
# extracted/: every path is forbidden except extracted/<v>/manifest.jsonl and manifest.sha1 (tracked, hashes only).
# <v> is any two-letter version folder (config/versions.tsv), so a new version's folder is covered too.
# fnmatchcase has no negation and `*` crosses `/`, so the exception is a divergence chain: each line forbids the paths
# that leave an allowed name at one position. Ceiling: a path that is a proper prefix of an allowed name
# (e.g. extracted/us/manifest.json) passes the path rule; the content-hash rule still applies to it.
glob:  extracted/[!a-z]*
